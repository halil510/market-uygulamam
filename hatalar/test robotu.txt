  RobotRapor({
    required this.baslangic,
    required this.bitis,
    required this.gezilenEkran,
    required this.toplamEkran,
    required this.tıklananWidget,
    required this.scrollSayisi,
    required this.yakalananHata,
    required this.bulgular,
    required this.ekranSureleriMs,
  });

  Duration get sure => bitis.difference(baslangic);

// Widget bulucu, tıklayıcı, doğrulayıcı yardımcılar.
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';

class RobotYardimcilar {
  final WidgetTester tester;
  RobotYardimcilar(this.tester);

  // ──────────────────────────────────────────────────────────────
  // YIKICI KONTROLÜ
  // ──────────────────────────────────────────────────────────────

  /// Widget yıkıcı mı? İçindeki metni ve ikon kodunu kontrol eder.
  bool yikiciMi(Widget widget) {
    // 1) Doğrudan Text
    if (widget is Text) {
      final t = (widget.data ?? '').toLowerCase();
      if (RobotAyarlari.yikiciMetinler.any((k) => t.contains(k))) return true;
    }
    // 2) İkon
    if (widget is Icon) {
      final kod = widget.icon?.codePoint;
      if (kod != null && RobotAyarlari.yikiciIkonKodlari.contains(kod)) {
        return true;
      }
    }
    // 3) İçinde çocuk var — recursive
    try {
      if (widget is ButtonStyleButton) {
        // Butonun child'ı Text ise kontrol et
        final child = widget.child;
        if (child is Text) {
          final t = (child.data ?? '').toLowerCase();
          if (RobotAyarlari.yikiciMetinler.any((k) => t.contains(k))) {
            return true;
          }
        }
      }
    } catch (_) {}
    return false;
  }

  /// Görünür mü ve ekran içinde mi?
  bool gorunurMu(Finder finder) {
    try {
      if (finder.evaluate().isEmpty) return false;
      final e = finder.evaluate().first;
      if (!e.mounted) return false;
      final konum = tester.getCenter(finder, warnIfMissed: false);
      final boyut = tester.view.physicalSize / tester.view.devicePixelRatio;
      return konum.dx >= 0 &&
          konum.dy >= 0 &&
          konum.dx <= boyut.width &&
          konum.dy <= boyut.height;
    } catch (_) {
      return false;
    }
  }

  // ──────────────────────────────────────────────────────────────
  // GÜVENLİ TIKLAMA

  /// Yıkıcı değilse tıklar. Tıkladıysa true döner.
  /// Dialog/bottom sheet açıldıysa otomatik kapatır.
  Future<bool> guvenliTikla(Finder finder) async {
    try {
      if (finder.evaluate().isEmpty) return false;
      if (!gorunurMu(finder)) return false;

      final widget = finder.evaluate().first.widget;
      if (yikiciMi(widget)) return false;

      await tester.tap(finder, warnIfMissed: false);
      await tester.pumpAndSettle(RobotAyarlari.tiklamaBekle);

      // Dialog/sheet açıldıysa kapat
      await _dialogKapat();
      return true;
    } catch (_) {
      return false;
    }
  }

  Future<void> _dialogKapat() async {
    for (var i = 0; i < 3; i++) {
      final dialogVar = find.byType(Dialog).evaluate().isNotEmpty;
      final sheetVar = find.byType(BottomSheet).evaluate().isNotEmpty;
      final popupVar = find.byType(PopupMenuItem).evaluate().isNotEmpty;
      if (!dialogVar && !sheetVar && !popupVar) return;

      try {
        // Escape benzeri: varsa "İptal"/"Kapat" butonuna bas, yoksa back
        final iptal = find.widgetWithText(TextButton, 'İptal');
        final kapat = find.widgetWithText(TextButton, 'Kapat');
        final vazgeç = find.widgetWithText(TextButton, 'Vazgeç');

        if (iptal.evaluate().isNotEmpty) {
          await tester.tap(iptal.first, warnIfMissed: false);
        } else if (kapat.evaluate().isNotEmpty) {
          await tester.tap(kapat.first, warnIfMissed: false);
        } else if (vazgeç.evaluate().isNotEmpty) {
          await tester.tap(vazgeç.first, warnIfMissed: false);
        } else {
          await tester.binding.handlePopRoute();
        }
        await tester.pumpAndSettle(RobotAyarlari.animasyonBekle);
      } catch (_) {
        break;
      }
    }
  }

  // ──────────────────────────────────────────────────────────────
  // EKRAN DOĞRULAMA
  // ──────────────────────────────────────────────────────────────

  /// Ekranın sağlıklı olup olmadığını kontrol eder.
  /// (rota, ekranSinifi) → listede dönen sorunlar.
  List<String> ekraniDogrula(String rota, EkranSinifi sinif) {
    final sorunlar = <String>[];

    // 1) ErrorWidget var mı?
    final errorVar = find
        .byWidgetPredicate((w) =>
            w.runtimeType.toString() == 'ErrorWidget' ||
            w.runtimeType.toString().contains('ErrorWidget'))
        .evaluate()
        .isNotEmpty;
    if (errorVar) sorunlar.add('Kırmızı ErrorWidget ekranda görünüyor');

    // 2) "Bir şeyler ters gitti" metni
    final hataText = find
        .byWidgetPredicate((w) =>
            w is Text &&
            RobotAyarlari.hataMetinleri.any(
                (m) => (w.data ?? '').toLowerCase().contains(m)))
        .evaluate()
        .isNotEmpty;
    if (hataText) sorunlar.add('Ekranda hata metni tespit edildi');

    // 3) Boş ekran — Scaffold var mı?
    if (find.byType(Scaffold).evaluate().isEmpty) {
      sorunlar.add('Scaffold bulunamadı (sayfa render edilmemiş olabilir)');
    }

    // 4) Salt-okunur ve kritik ekranlarda minimum içerik beklentisi
    if (sinif == EkranSinifi.saltOkunur || sinif == EkranSinifi.kritik) {
      final textSayisi = find.byType(Text).evaluate().length;
      if (textSayisi < 3) {
        sorunlar.add('Salt-okunur ekranda çok az içerik (Text=$textSayisi)');
      }
    }

    // 5) Form ekranlarında en az bir input alanı beklenir
    if (sinif == EkranSinifi.form) {
      final inputVar = find.byType(TextField).evaluate().isNotEmpty ||
          find.byType(TextFormField).evaluate().isNotEmpty ||
          find.byType(DropdownButtonFormField).evaluate().isNotEmpty;
      if (!inputVar) {
        sorunlar.add('Form ekranı ama hiç input alanı bulunamadı');
      }
    }

    // 6) Liste ekranlarında en az bir liste / grid beklenir
    if (sinif == EkranSinifi.liste) {
      final listeVar = find.byType(ListView).evaluate().isNotEmpty ||
          find.byType(GridView).evaluate().isNotEmpty ||
          find.byType(SingleChildScrollView).evaluate().isNotEmpty;
      if (!listeVar) {
        sorunlar.add('Liste ekranı ama ListView/GridView yok');
      }
    }

    return sorunlar;
  }

  // ──────────────────────────────────────────────────────────────
  // TIKLANABİLİR WIDGET TOPLAMA
  // ──────────────────────────────────────────────────────────────

  /// Ekrandaki tüm tıklanabilir widget'ları güvenli sırayla toplar.
  /// Yıkıcı olanları FİLTRELER.
  List<Finder> tiklanabilirleriTopla({int limit = 30}) {
    final liste = <Finder>[];
    final bulundu = <int>{};

    void ekle(Finder f) {
      if (liste.length >= limit) return;
      try {
        for (final e in f.evaluate()) {
          if (!e.mounted) continue;
          final hashCode = e.widget.hashCode;
          if (bulundu.contains(hashCode)) continue;
          if (yikiciMi(e.widget)) continue;
          bulundu.add(hashCode);
          liste.add(find.byWidget(e.widget));
          if (liste.length >= limit) break;
        }
      } catch (_) {}
    }

    // Önce spesifik widget'lar (daha güvenli), sonra genel.
    ekle(find.byType(Card));
    ekle(find.byType(ListTile));
    ekle(find.byType(FilledButton));
    ekle(find.byType.OutlinedButton);
    ekle(find.byType.ElevatedButton);
    ekle(find.byType(TextButton));
    ekle(find.byType(ChoiceChip));
    ekle(find.byType(FilterChip));
    ekle(find.byType(ActionChip));
    ekle(find.byType(Tab));
    ekle(find.byType(InkWell));

    return liste;
  }

  /// Ekranda scroll yapılabilir alan varsa yukarı-aşağı kaydırır.

        sayi++;
      }
      // Başa dön
      for (var i = 0; i < RobotAyarlari.maxScrollDenemesi; i++) {
        if (scrollables.evaluate().isEmpty) break;
        await tester.drag(scrollables.first, const Offset(0, 250));
        await tester.pumpAndSettle(RobotAyarlari.animasyonBekle);
      }
    } catch (_) {}
    return sayi;
  }

  // SEKME GEZME
  // ──────────────────────────────────────────────────────────────

  /// Ekranda TabBar varsa her sekmeye sırayla geçer.
  Future<int> sekmeleriGez() async {
    var sayi = 0;
    try {
      final tabBar = find.byType(TabBar);
      if (tabBar.evaluate().isEmpty) return 0;

      final tabs = find.descendant(of: tabBar, matching: find.byType(Tab));
      final tabSayisi = tabs.evaluate().length;
      if (tabSayisi <= 1) return 0;

      for (var i = 0; i < tabSayisi; i++) {
        try {
          final tab = tabs.at(i);
          if (!gorunurMu(tab)) continue;
          await tester.tap(tab, warnIfMissed: false);
          await tester.pumpAndSettle(RobotAyarlari.animasyonBekle);
          sayi++;
        } catch (_) {}
      }
      // İlk sekmeye dön
      try {
        await tester.tap(tabs.first, warnIfMissed: false);
        await tester.pumpAndSettle(RobotAyarlari.animasyonBekle);
      } catch (_) {}
    } catch (_) {}
    return sayi;
  }

        deneme++;

    }
  }

  // ═══════════════════════════════════════════════════════════════
  // GÜVENLİ TIKLAMA DÖNGÜSÜ
  // ═══════════════════════════════════════════════════════════════
  Future<void> _guvenliTiklaDongusu(String rota, EkranSinifi sinif) async {
    final adaylar = _yardimci.tiklanabilirleriTopla(
      limit: RobotAyarlari.maxTiklamaSayisi,
    );
    if (adaylar.isEmpty) {
      _bulgu(rota, sinif.name, BulguTipi.bilgi,
          'Tıklanabilir güvenli widget bulunamadı');
      return;
    }

    for (var i = 0; i < adaylar.length; i++) {
      // Her tıklama öncesi ana ekrana dön (bazı öğeler navigasyon yapabilir)
      if (!_aktifRotaKontrol(rota)) {
        await _geriDon();
        await _rotaDegistir(rota);
      }

      final tıkladı = await _yardimci.guvenliTikla(adaylar[i]);
      if (tıkladı) {
        _tıklananWidget++;
        // Anlık hata oluştu mu?
        if (_anlikHatalar.isNotEmpty) {
          for (final h in _anlikHatalar) {
            _bulgu(rota, sinif.name, BulguTipi.hata,
                'Tıklama sonrası: $h');
          }
          _anlikHatalar.clear();
        }
      }
    }
  }

  bool _aktifRotaKontrol(String rota) {
    try {
      final scaffold = find.byType(Scaffold);
      if (scaffold.evaluate().isEmpty) return false;
      final ctx = tester.element(scaffold.first);
      final currentRota = GoRouterState.of(ctx).uri.toString();
      return currentRota == rota;
    } catch (_) {
      return false;
    }
  }

  // ═══════════════════════════════════════════════════════════════
  // GERİ DÖN
  // ═══════════════════════════════════════════════════════════════
  Future<void> _geriDon() async {
    try {
      // Açık dialog/sheet varsa kapat
      for (var i = 0; i < 3; i++) {
        final dialogVar = find.byType(Dialog).evaluate().isNotEmpty;
        final sheetVar = find.byType(BottomSheet).evaluate().isNotEmpty;
        if (!dialogVar && !sheetVar) break;
        await tester.binding.handlePopRoute();
        await tester.pumpAndSettle(RobotAyarlari.animasyonBekle);
      }
      // Ana sayfaya go_router ile dön
      final scaffold = find.byType(Scaffold);
      if (scaffold.evaluate().isNotEmpty) {
        final ctx = tester.element(scaffold.first);
        GoRouter.of(ctx).go('/');
        await tester.pumpAndSettle(RobotAyarlari.sayfaGecisBekle);
      }
    } catch (_) {}
  }

  // ═══════════════════════════════════════════════════════════════
  // BULGU EKLE
  // ═══════════════════════════════════════════════════════════════
  void _bulgu(
    String ekran,
    String sinif,
    BulguTipi tip,
    String mesaj, {
    String? ayrinti,
    String? stack,
  }) {
    _bulgular.add(RobotBulgusu(
      ekran: ekran,
      ekranSinifi: sinif,
      tip: tip,
      mesaj: mesaj,
      ayrinti: ayrinti,
      stack: stack,
    ));
    final ikon = switch (tip) {
      BulguTipi.hata => '❌',
      BulguTipi.uyari => '⚠️ ',
      BulguTipi.bilgi => 'ℹ️ ',
      BulguTipi.basarili => '✅',
    };
    _log('│  $ikon [$ekran] $mesaj');
  }

  void _log(String m) {
    // ignore: avoid_print
    print('[ROBOT] $m');
  }
}
